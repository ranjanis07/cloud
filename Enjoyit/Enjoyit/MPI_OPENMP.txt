>>sum of 100 numbers(Program)

#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[])
{
    int rank, size;
    int i, local_sum = 0, total_sum = 0;

    // Initialize MPI
    MPI_Init(&argc, &argv);

    // Get process rank and total processes
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    // Calculate partial sum
    for (i = rank + 1; i <= 100; i += size)
    {
        local_sum += i;
    }

    printf("Process %d: Local Sum = %d\n", rank, local_sum);

    // Combine partial sums
    MPI_Reduce(&local_sum, &total_sum, 1, MPI_INT,
               MPI_SUM, 0, MPI_COMM_WORLD);

    // Print the final result
    if (rank == 0)
    {
        printf("Sum of numbers from 1 to 100 = %d\n", total_sum);
    }

    // End MPI
    MPI_Finalize();

    return 0;
}

>>to compile and run

mpicc sum100.c -o sum100

mpirun --oversubscribe -np 4 ./sum100

#include <stdio.h>
#include <omp.h>

int main()
{
    int i, sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (i = 1; i <= 100; i++)
    {
        sum += i;
    }

    printf("Sum of numbers from 1 to 100 = %d\n", sum);

    return 0;
}
