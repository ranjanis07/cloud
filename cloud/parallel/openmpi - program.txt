#include <stdio.h>
#include <omp.h>

int main()
{
    int arr[100], even[100], j = 0;

    for (int i = 0; i < 100; i++)
        arr[i] = i + 1;

    #pragma omp parallel for
    for (int i = 0; i < 100; i++)
    {
        if (arr[i] % 2 == 0)
        {
            #pragma omp critical
            {
                even[j++] = arr[i];
            }
        }
    }

    printf("Even numbers:\n");

    for (int i = 0; i < j; i++)
        printf("%d ", even[i]);

    return 0;
}
